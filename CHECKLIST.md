**Setup Example:**
```bash
git clone <repo>
cd fhevm-sdk
pnpm install
cd packages/nextjs
pnpm dev
```

**Usage Example:**
```tsx
import { useFhevm, useFHEEncryption } from '@fhevm-sdk/react';
const { instance } = useFhevm({ provider, chainId });
const { encryptWith } = useFHEEncryption({ instance, ethersSigner, contractAddress });
```
